# Why MongoDB Over PostgreSQL

| Factor | PostgreSQL | MongoDB |
|--------|-----------|---------|
| Schema Flexibility | Rigid tables | Dynamic documents |
| FIR Structure | 15+ JOINs per query | Single document read |
| Nested Data (victims, accused, acts) | Separate tables + JOINs | Embedded arrays |
| Varying Fields Per Crime Type | NULL columns everywhere | Only store what exists |
| Evolution | ALTER TABLE migrations | Add fields organically |
| Geographic Queries | PostGIS (good) | GeoJSON native (great) |
| Aggregation Pipeline | SQL GROUP BY | Powerful aggregation framework |
| JSON Export | Manual casting | Native BSON/JSON |

**Key Insight**: An FIR is inherently a **document** — it contains nested arrays of victims, accused, act-sections, arrests. MongoDB's document model maps this naturally without 15-way JOINs.

# MongoDB Database Design

## Design Philosophy

```text
Embedding Strategy:
├── Transactional data WITH the case document
│   ├── victims[]
│   ├── accused[]
│   ├── complainants[]
│   ├── actSections[]
│   ├── arrestSurrenders[]
│   └── chargesheet
│
Referencing Strategy:
├── Master/reference data as separate collections
│   ├── units (police stations)
│   ├── employees
│   ├── courts
│   ├── districts
│   ├── states
│   ├── acts & sections
│   └── crime heads
│
Denormalization (for read performance):
├── Duplicate frequently queried fields
│   ├── districtName in caseMaster
│   ├── unitName in caseMaster
│   └── crimeHeadName in caseMaster
```

### Indexes for `caseMaster`

```javascript
// Primary lookup — Crime Number
db.caseMaster.createIndex(
  { crimeNo: 1 },
  { unique: true }
)

// Case Number lookup
db.caseMaster.createIndex(
  { caseNo: 1, policeStationID: 1 }
)

// Date range queries (most common)
db.caseMaster.createIndex(
  { crimeRegisteredDate: -1 }
)

// District + Date (dashboard filters)
db.caseMaster.createIndex(
  { districtID: 1, crimeRegisteredDate: -1 }
)

// Station + Date (station-level reports)
db.caseMaster.createIndex(
  { policeStationID: 1, crimeRegisteredDate: -1 }
)

// Crime head analysis
db.caseMaster.createIndex(
  { crimeMajorHeadID: 1, crimeRegisteredDate: -1 }
)

// Status tracking
db.caseMaster.createIndex(
  { caseStatusID: 1, districtID: 1 }
)

// GEOGRAPHIC — Near queries for hotspots
db.caseMaster.createIndex(
  { location: "2dsphere" }
)

// TEXT SEARCH — Full-text on narrative
db.caseMaster.createIndex(
  {
    briefFacts: "text",
    "accused.accusedName": "text",
    "victims.victimName": "text",
    "complainants.complainantName": "text"
  },
  {
    weights: {
      briefFacts: 10,
      "accused.accusedName": 5,
      "victims.victimName": 5,
      "complainants.complainantName": 3
    },
    name: "case_text_search"
  }
)

// Accused name search (embedded array)
db.caseMaster.createIndex(
  { "accused.accusedName": 1 }
)

// Act-Section queries (e.g., all IPC 302 cases)
db.caseMaster.createIndex(
  { "actSections.actCode": 1, "actSections.sectionCode": 1 }
)

// Gravity + District (risk analysis)
db.caseMaster.createIndex(
  { gravityOffenceID: 1, districtID: 1, crimeRegisteredDate: -1 }
)

// Compound: Unit + Status + Date
db.caseMaster.createIndex(
  { policeStationID: 1, caseStatusID: 1, crimeRegisteredDate: -1 }
)
```

## Graph Sync from MongoDB (Python)

```python
from neo4j import GraphDatabase
from pymongo import MongoClient

def sync_case_to_graph(mongo_doc):
    """Sync a single caseMaster document to Neo4j."""
    with neo4j_driver.session() as session:
        # Create Case node
        session.run("""
            MERGE (c:Case {crimeNo: $crimeNo})
            SET c.caseMasterID = $caseMasterID,
                c.crimeType = $crimeType,
                c.crimeDate = $crimeDate,
                c.status = $status,
                c.district = $district
        """, crimeNo=mongo_doc["crimeNo"],
             caseMasterID=mongo_doc["caseMasterID"],
             crimeType=mongo_doc["crimeMinorHeadName"],
             crimeDate=mongo_doc["crimeRegisteredDate"],
             status=mongo_doc["caseStatusName"],
             district=mongo_doc["districtName"])

        # Create Person nodes and relationships
        for accused in mongo_doc.get("accused", []):
            session.run("""
                MERGE (p:Person {name: $name, personID: $pid})
                SET p.role = 'accused', p.age = $age, p.gender = $gender
                WITH p
                MATCH (c:Case {crimeNo: $crimeNo})
                MERGE (p)-[:ACCUSED_IN]->(c)
            """, name=accused["accusedName"],
                 pid=accused["personID"],
                 age=accused["ageYear"],
                 gender=accused["genderID"],
                 crimeNo=mongo_doc["crimeNo"])

        # Create co-accused associations
        accused_names = [a["accusedName"] for a in mongo_doc.get("accused", [])]
        for i, name1 in enumerate(accused_names):
            for name2 in accused_names[i+1:]:
                session.run("""
                    MATCH (p1:Person {name: $name1})
                    MATCH (p2:Person {name: $name2})
                    MERGE (p1)-[:ASSOCIATED_WITH {case: $crimeNo}]->(p2)
                """, name1=name1, name2=name2, crimeNo=mongo_doc["crimeNo"])

        # Act-Section relationships
        for act_sec in mongo_doc.get("actSections", []):
            session.run("""
                MERGE (s:ActSection {actCode: $act, sectionCode: $sec})
                SET s.description = $desc
                WITH s
                MATCH (c:Case {crimeNo: $crimeNo})
                MERGE (c)-[:CHARGED_UNDER]->(s)
            """, act=act_sec["actCode"],
                 sec=act_sec["sectionCode"],
                 desc=act_sec["sectionDescription"],
                 crimeNo=mongo_doc["crimeNo"])
```

```python
from sentence_transformers import SentenceTransformer

model = SentenceTransformer('all-MiniLM-L6-v2')

def semantic_search(query: str, district: str = None, top_k: int = 10):
    """Search FIRs by meaning, not just keywords."""

    query_vector = model.encode(query).tolist()

    # Build filters
    conditions = []
    if district:
        conditions.append(
            FieldCondition(key="district", match=MatchValue(value=district))
        )

    results = qdrant.search(
        collection_name="fir_embeddings",
        query_vector=query_vector,
        query_filter=Filter(must=conditions) if conditions else None,
        limit=top_k,
        with_payload=True
    )

    return [
        {
            "crime_no": r.payload["crime_no"],
            "crime_type": r.payload["crime_type"],
            "district": r.payload["district"],
            "register_date": r.payload["register_date"],
            "status": r.payload["status"],
            "snippet": r.payload["brief_facts_snippet"],
            "similarity_score": r.score
        }
        for r in results
    ]

# Example: "Night time house break-ins where locks were broken"
results = semantic_search(
    "Night time house break-ins where locks were broken",
    district="Bengaluru Urban"
)
```

## Similar Case Discovery

```python
def find_similar_cases(crime_no: str, top_k: int = 5):
    """Find cases similar to a given case."""

    # Get the vector for the reference case
    points = qdrant.scroll(
        collection_name="fir_embeddings",
        scroll_filter=Filter(must=[
            FieldCondition(key="crime_no", match=MatchValue(value=crime_no))
        ]),
        limit=1
    )

    if not points[0]:
        return []

    reference_vector = points[0][0].vector

    # Search for similar vectors (excluding the reference itself)
    results = qdrant.search(
        collection_name="fir_embeddings",
        query_vector=reference_vector,
        query_filter=Filter(must_not=[
            FieldCondition(key="crime_no", match=MatchValue(value=crime_no))
        ]),
        limit=top_k,
        with_payload=True
    )

    return results
```

# RAG Pipeline — MongoDB + Qdrant + Gemini

```python
from langchain_google_genai import ChatGoogleGenerativeAI
from langchain.prompts import ChatPromptTemplate
from pymongo import MongoClient

llm = ChatGoogleGenerativeAI(model="gemini-2.5-pro", temperature=0.1)

def answer_question(question: str, district: str = None) -> dict:
    """
    Full RAG pipeline:
    1. Semantic search in Qdrant
    2. Fetch full documents from MongoDB
    3. Send context + question to Gemini
    4. Return evidence-based answer
    """

    # Step 1: Semantic search
    relevant_hits = semantic_search(question, district=district, top_k=5)

    if not relevant_hits:
        return {
            "answer": "No relevant cases found for your query.",
            "evidence": [],
            "sources": []
        }

    # Step 2: Fetch full documents from MongoDB
    crime_nos = [hit["crime_no"] for hit in relevant_hits]
    cases = list(mongo_db.caseMaster.find(
        {"crimeNo": {"$in": crime_nos}, "isActive": True},
        {
            "crimeNo": 1, "caseNo": 1, "crimeMinorHeadName": 1,
            "crimeRegisteredDate": 1, "districtName": 1,
            "policeStationName": 1, "caseStatusName": 1,

            "arrestSurrenders.arrestSurrenderTypeName": 1,
            "chargesheet.csDate": 1, "chargesheet.csTypeName": 1

    ))

    # Step 3: Build context for LLM
    context_parts = []
    for case in cases:
```

# Embedding Sync Pipeline

**Background worker that keeps Qdrant in sync with MongoDB.**

```python
from sentence_transformers import SentenceTransformer
from datetime import datetime
import hashlib

model = SentenceTransformer('all-MiniLM-L6-v2')

def generate_fir_embedding(case_doc: dict) -> list:
    """Generate embedding from FIR narrative + metadata."""

    # Combine narrative with structured metadata for richer embeddings
    text_parts = [
        case_doc.get("briefFacts", ""),
        f"Crime type: {case_doc.get('crimeMinorHeadName', '')}",
        f"District: {case_doc.get('districtName', '')}",
        f"Station: {case_doc.get('policeStationName', '')}",
    ]

    # Add accused names
    for a in case_doc.get("accused", []):
        text_parts.append(f"Accused: {a['accusedName']}")

    # Add victim names
    for v in case_doc.get("victims", []):
        text_parts.append(f"Victim: {v['victimName']}")

    # Add act-sections
    for asoc in case_doc.get("actSections", []):
        text_parts.append(
            f"{asoc['actCode']} {asoc['sectionCode']}: {asoc['sectionDescription']}"
        )

    combined_text = " ".join(text_parts)
    return model.encode(combined_text).tolist()


def get_content_hash(case_doc: dict) -> str:
    """Hash of fields that affect embedding — skip re-embedding if unchanged."""
    content = {
        "briefFacts": case_doc.get("briefFacts", ""),
        "crimeMinorHeadName": case_doc.get("crimeMinorHeadName", ""),
        "accused": [a["accusedName"] for a in case_doc.get("accused", [])],
        "victims": [v["victimName"] for v in case_doc.get("victims", [])],
        "actSections": [
            f"{a['actCode']} {a['sectionCode']}"
            for a in case_doc.get("actSections", [])
        ]
    }
    return hashlib.md5(str(content).encode()).hexdigest()


def sync_unembedded_cases():
    """Find cases without embeddings and generate them."""

    # Find cases not yet embedded
    unembedded = mongo_db.caseMaster.find(
        {
            "embeddingGenerated": {"$ne": True},
            "isActive": True
        },
        { "_id": 0, "caseMasterID": 1 }
    ).limit(100)

    for ref in unembedded:
        case = mongo_db.caseMaster.find_one(
            {"caseMasterID": ref["caseMasterID"]},
            { "_id": 0 }
        )

        if not case:
            continue

        vector = generate_fir_embedding(case)
        content_hash = get_content_hash(case)

        # Upsert to Qdrant
        qdrant.upsert(
            collection_name="fir_embeddings",
            points=[PointStruct(
                id=case["caseMasterID"],
                vector=vector,
                payload={
                    "case_master_id": case["caseMasterID"],
                    "crime_no": case["crimeNo"],
                    "crime_type": case.get("crimeMinorHeadName", ""),
                    "crime_head": case.get("crimeMajorHeadName", ""),
                    "district": case.get("districtName", ""),
                    "station": case.get("policeStationName", ""),
                    "register_date": case["crimeRegisteredDate"].isoformat(),
                    "status": case.get("caseStatusName", ""),
                    "gravity": case.get("gravityOffenceName", ""),
                    "acts_sections": [
                        f"{a['actCode']} {a['sectionCode']}"
                        for a in case.get("actSections", [])
                    ],
                    "accused_names": [
                        a["accusedName"] for a in case.get("accused", [])
                    ],
                    "victim_names": [
                        v["victimName"] for v in case.get("victims", [])
                    ],
                    "brief_facts_snippet": case.get("briefFacts", "")[:300],
                    "content_hash": content_hash
                }
            )]
        )

        # Mark as embedded in MongoDB
        mongo_db.caseMaster.update_one(
            {"caseMasterID": case["caseMasterID"]},
            {
                "$set": {
                    "embeddingGenerated": True,
                    "contentHash": content_hash,
                    "updatedAt": datetime.utcnow()
                }
            }
        )
```

```python
from sklearn.cluster import DBSCAN
import numpy as np

def detect_hotspots(
    district_id: int,
    from_date: datetime,
    to_date: datetime,
    crime_head_id: int = None,
    eps_meters: float = 500,               // 500m cluster radius
    min_cases: int = 5                      // Minimum cases to form hotspot
):
    """
    DBSCAN clustering on MongoDB geo data.
    Returns hotspot clusters for map visualization.
    """

    # Fetch coordinates from MongoDB
    query = {
        "districtID": district_id,
        "crimeRegisteredDate": {"$gte": from_date, "$lt": to_date},
        "location": {"$exists": True},
        "isActive": True
    }
    if crime_head_id:
        query["crimeMajorHeadID"] = crime_head_id

    cases = mongo_db.caseMaster.find(
        query,
        {
            "crimeNo": 1,
            "location.coordinates": 1,
            "crimeMinorHeadName": 1,
            "crimeRegisteredDate": 1
        }
    )

    # Extract coordinates
    coords = []
    metadata = []
    for c in cases:
        lon, lat = c["location"]["coordinates"]
        coords.append([lat, lon])          // DBSCAN expects [lat, lon]
        metadata.append(c)

    if len(coords) < min_cases:
        return {"hotspots": [], "total_cases": len(coords)}

    # Convert meters to degrees (approximate)
    eps_deg = eps_meters / 111320           // 1 degree ≈ 111.32 km

    # Run DBSCAN
    clustering = DBSCAN(
        eps=eps_deg,
        min_samples=min_cases,
        metric='haversine',
        algorithm='ball_tree'
    ).fit(np.radians(coords))

    # Build hotspot results
    hotspots = []
    labels = clustering.labels_
    unique_labels = set(labels) - {-1}      // Exclude noise points

    for label in unique_labels:
        cluster_indices = np.where(labels == label)[0]
        cluster_coords = np.array(coords)[cluster_indices]
        center_lat = cluster_coords[:, 0].mean()
        center_lon = cluster_coords[:, 1].mean()

        # Crime type distribution in cluster
        crime_types = {}
        for idx in cluster_indices:
            ct = metadata[idx].get("crimeMinorHeadName", "Unknown")
            crime_types[ct] = crime_types.get(ct, 0) + 1

        hotspots.append({
            "hotspot_id": int(label),
            "center": {
                "type": "Point",
                "coordinates": [float(center_lon), float(center_lat)]
            },
            "case_count": int(len(cluster_indices)),
            "radius_meters": int(eps_meters),
            "crime_type_distribution": crime_types,
            "top_crime_type": max(crime_types, key=crime_types.get),
            "cases": [
                {
                    "crime_no": metadata[idx]["crimeNo"],
                    "type": metadata[idx].get("crimeMinorHeadName"),
                    "date": metadata[idx]["crimeRegisteredDate"].isoformat()
                }
                for idx in cluster_indices
            ]
        })

    # Sort by case count
    hotspots.sort(key=lambda h: h["case_count"], reverse=True)

    return {
        "hotspots": hotspots,
        "total_cases": len(coords),
        "noise_cases": int(np.sum(labels == -1)),
        "parameters": {
            "district_id": district_id,
            "eps_meters": eps_meters,
            "min_cases": min_cases,
            "date_range": {
                "from": from_date.isoformat(),
                "to": to_date.isoformat()
            }
        }
    }
```

```python
import xgboost as xgb
import pandas as pd
import shap

def build_prediction_features(district_id: int, months_back: int = 24) -> pd.DataFrame:
    """
    Extract time-series features from MongoDB for crime prediction.
    """

    pipeline = [
        {
            "$match": {
                "districtID": district_id,
                "crimeRegisteredDate": {
                    "$gte": datetime.utcnow() - timedelta(days=months_back * 30)
                },
                "isActive": True
            }
        },
        {
            "$group": {
                "_id": {
                    "year": { "$year": "$crimeRegisteredDate" },
                    "month": { "$month": "$crimeRegisteredDate" },
                    "crimeHead": "$crimeMajorHeadID"
                },
                "count": { "$sum": 1 },
                "heinous": {
                    "$sum": { "$cond": [{ "$eq": ["$gravityOffenceID", 1] }, 1, 0] }
                },
                "chargeSheeted": {
                    "$sum": { "$cond": [{ "$eq": ["$caseStatusID", 3] }, 1, 0] }
                },
                "arrests": { "$sum": { "$size": "$arrestSurrenders" } }
            }
        },
        { "$sort": { "_id.year": 1, "_id.month": 1, "_id.crimeHead": 1 } }
    ]

    monthly_data = list(mongo_db.caseMaster.aggregate(pipeline))

    # Build feature DataFrame
    rows = []
    for entry in monthly_data:
        rows.append({
            "year": entry["_id"]["year"],
            "month": entry["_id"]["month"],
            "crime_head_id": entry["_id"]["crimeHead"],
            "total_cases": entry["count"],
            "heinous_cases": entry["heinous"],
            "charge_sheeted": entry["chargeSheeted"],
            "arrests": entry["arrests"]
        })

    df = pd.DataFrame(rows)

    # Add lag features (previous month, same month last year)
    df = df.sort_values(["crime_head_id", "year", "month"])
    df["prev_month_cases"] = df.groupby("crime_head_id")["total_cases"].shift(1)
    df["same_month_last_year"] = df.groupby("crime_head_id")["total_cases"].shift(12)
    df["month_over_month_change"] = (
        df["total_cases"] - df["prev_month_cases"]
    ) / df["prev_month_cases"].replace(0, 1)

    # Add time features
    df["is_festive_season"] = df["month"].isin([10, 11, 12, 1, 3, 4]).astype(int)
    df["is_monsoon"] = df["month"].isin([6, 7, 8, 9]).astype(int)

    return df.dropna()


def train_predictor(district_id: int):
    """Train XGBoost model and return predictions with SHAP explanations."""

    df = build_prediction_features(district_id)

    feature_cols = [
        "prev_month_cases", "same_month_last_year",
        "month_over_month_change", "heinous_cases",
        "charge_sheeted", "arrests",
        "is_festive_season", "is_monsoon"
    ]

    X = df[feature_cols]
    y = df["total_cases"]

    # Train
    model = xgb.XGBRegressor(
        n_estimators=100,
        max_depth=4,
        learning_rate=0.1,
        random_state=42
    )
    model.fit(X, y)

    # Predict next month
    latest = df[df.groupby("crime_head_id")["year"].rank(
        method="dense", ascending=False
    ) == 1]

    predictions = []
    explainer = shap.TreeExplainer(model)

    for _, row in latest.iterrows():
        features = row[feature_cols].values.reshape(1, -1)
        pred = model.predict(features)[0]

        # SHAP explanation
        shap_values = explainer.shap_values(features)
        top_factors = sorted(
            zip(feature_cols, shap_values[0]),
            key=lambda x: abs(x[1]),
            reverse=True
        )[:3]

        predictions.append({
            "crime_head_id": int(row["crime_head_id"]),
            "predicted_cases_next_month": round(float(pred), 1),
            "previous_month_actual": int(row["total_cases"]),
            "change_percent": round(
                ((pred - row["total_cases"]) / row["total_cases"]) * 100, 1
            ),
            "risk_level": (
                "HIGH" if pred > row["total_cases"] * 1.2
                else "MEDIUM" if pred > row["total_cases"] * 1.05
                else "LOW"
            ),
            "key_factors": [
                {"feature": f, "impact": round(float(v), 2)}
                for f, v in top_factors
            ]
        })

    predictions.sort(key=lambda p: p["predicted_cases_next_month"], reverse=True)

    return {
        "district_id": district_id,
        "prediction_month": "next",
        "predictions": predictions,
        "model_performance": {
            "training_samples": len(df),
            "features_used": feature_cols
        }
    }
```

# Entity Resolution — Fuzzy Matching Across Cases

```python
from rapidfuzz import fuzz, process

def find_duplicate_accused(
    name: str,
    threshold: float = 85,
    max_results: int = 10
):
    """
    Find potential duplicate accused across all cases using fuzzy matching.
    Uses MongoDB text index + rapidfuzz for scoring.
    """

    # Step 1: Coarse filter using MongoDB text search
    # Split name into tokens for broader search
    tokens = name.split()
    text_query = " ".join(tokens[:2])       // Use first 2 name parts

    candidates = mongo_db.caseMaster.find(
        {
            "accused.accusedName": { "$regex": tokens[0], "$options": "i" },
            "isActive": True
        },
        {
            "crimeNo": 1,
            "districtName": 1,
            "accused.accusedName": 1,
            "accused.accusedMasterID": 1,
            "accused.personID": 1,
            "accused.ageYear": 1,
            "accused.genderID": 1
        }
    ).limit(200)

    # Step 2: Fine-grained fuzzy matching
    matches = []
    seen_ids = set()

    for case in candidates:
        for accused in case.get("accused", []):
            if accused["accusedMasterID"] in seen_ids:
                continue
            seen_ids.add(accused["accusedMasterID"])

            # Multiple fuzzy matching strategies
            ratio = fuzz.ratio(name, accused["accusedName"])
            partial = fuzz.partial_ratio(name, accused["accusedName"])
            token_sort = fuzz.token_sort_ratio(name, accused["accusedName"])
            token_set = fuzz.token_set_ratio(name, accused["accusedName"])

            # Best score across strategies
            best_score = max(ratio, partial, token_sort, token_set)

            if best_score >= threshold:
                matches.append({
                    "matched_name": accused["accusedName"],
                    "score": best_score,
                    "match_type": (
                        "exact" if ratio == 100
                        else "partial" if partial == best_score
                        else "token_sort" if token_sort == best_score
                        else "token_set"
                    ),
                    "crime_no": case["crimeNo"],
                    "district": case["districtName"],
                    "age": accused.get("ageYear"),
                    "gender": accused.get("genderID"),
                    "person_id": accused.get("personID"),
                    "accused_master_id": accused["accusedMasterID"]
                })

    # Sort by best score
    matches.sort(key=lambda m: m["score"], reverse=True)
    return matches[:max_results]
```
